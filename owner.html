async function loadOrders() { const table = document.getElementById("orders-table"); if (!table) { console.error("Could not find orders-table."); return; } const tbody = table.querySelector("tbody") || table; tbody.innerHTML = ` Loading customer orders... `; try { const { data: orders, error } = await db .from("orders") .select("*") .order("created_at", { ascending: false }) .limit(300); if (error) throw error; if (!orders || orders.length === 0) { tbody.innerHTML = ` No customer orders have been placed yet. `; return; } const { data: items, error: itemsError } = await db .from("order_items") .select("order_id,product_name,color,size,quantity,unit_price,line_total"); if (itemsError) throw itemsError; const safe = value => String(value ?? "").replace(/[&<>"']/g, char => ({ "&": "&", "<": "<", ">": ">", '"': """, "'": "'" })[char]); const money = value => "KSh " + Number(value || 0).toLocaleString("en-KE", { maximumFractionDigits: 2 }); const normalStatus = value => String(value || "pending").toLowerCase(); tbody.innerHTML = ""; for (const order of orders) { const orderItems = (items || []).filter(item => item.order_id === order.id ); const itemText = orderItems.length ? orderItems.map(item => `${item.product_name} — ${item.color || "Colour not specified"}, ` + `${item.size || "Size not specified"} × ${item.quantity} ` + `(${money(item.line_total ?? item.unit_price * item.quantity)})` ).join("
") : "No item details found"; const fulfilment = order.fulfillment_type || order.customer_fulfillment_type || order.delivery_method || "delivery"; const address = [ order.delivery_address, order.delivery_location, order.town, order.county, order.delivery_area ].filter((value, index, array) => value && array.indexOf(value) === index ).join(", "); const date = order.schedule_date || order.delivery_date || "Not scheduled"; const time = order.schedule_slot || order.delivery_time || order.fulfillment_time || "Not specified"; const orderStatus = normalStatus(order.order_status); const paymentStatus = normalStatus(order.payment_status); const row = document.createElement("tr"); row.innerHTML = ` ${safe(String(order.id).slice(0, 8).toUpperCase())}
${safe(order.created_at ? new Date(order.created_at).toLocaleString() : "")} ${safe(order.customer_name || "Customer")}
${safe(order.customer_phone_number || order.customer_phone || "")}
${safe(order.customer_email || "")} ${itemText} ${safe(fulfilment)}
${safe(address || "No address supplied")}
${safe(date)} · ${safe(time)}
Delivery fee: ${money(order.delivery_fee)} ${order.order_notes ? `
Notes: ${safe(order.order_notes)}` : ""} Subtotal: ${money(order.subtotal)}
Discount: ${money(order.discount_amount)}
Total: ${money(order.total)} ${safe(order.payment_method || "Not specified")}
Payment: ${safe(paymentStatus)} ${order.mpesa_phone_number ? `
M-Pesa: ${safe(order.mpesa_phone_number)}` : ""} `; const orderSelect = row.querySelector(".owner-order-status"); const paymentSelect = row.querySelector(".owner-payment-status"); const saveButton = row.querySelector(".owner-save-order"); const statusMessage = row.querySelector(".owner-order-message"); saveButton.addEventListener("click", async () => { const newOrderStatus = orderSelect.value; const newPaymentStatus = paymentSelect.value; const paymentMethod = String(order.payment_method || "").toLowerCase(); if ( newPaymentStatus === "successful" && ["cash_on_delivery", "cash on delivery", "cod", "cash", "cash_at_pickup"].includes(paymentMethod) && !["paid", "success", "successful"].includes(paymentStatus) ) { const confirmed = window.confirm( "Confirm that you have actually received the customer's cash before marking this payment successful." ); if (!confirmed) { paymentSelect.value = "pending"; return; } } if ( newPaymentStatus === "successful" && ["mpesa", "m-pesa", "demo_mpesa"].includes(paymentMethod) && !["paid", "success", "successful"].includes(paymentStatus) ) { const confirmed = window.confirm( "Only mark M-Pesa successful after checking the actual payment in your M-Pesa account or transaction records. Have you verified receipt?" ); if (!confirmed) { paymentSelect.value = "pending"; return; } } saveButton.disabled = true; statusMessage.textContent = "Saving..."; try { const update = { order_status: newOrderStatus, payment_status: newPaymentStatus }; const { error: updateError } = await db .from("orders") .update(update) .eq("id", order.id); if (updateError) throw updateError; statusMessage.textContent = "Saved."; await loadOrders(); } catch (error) { console.error("Order update error:", error); statusMessage.textContent = "Could not save: " + error.message; saveButton.disabled = false; } }); tbody.appendChild(row); } } catch (error) { console.error("Loading orders failed:", error); tbody.innerHTML = ` Could not load orders: ${String(error.message || error)} `; } }
